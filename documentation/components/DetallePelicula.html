import { Component } from '@angular/core';
import { ActivatedRoute, RouterLink, Router } from '@angular/router';
import { PeliculaService } from '../../pelicula-service';
import { Pelicula } from '../../../../core/models/pelicula.model';
import { FuncionService } from '../../../funciones/funcion.service';
import { Funcion } from '../../../../core/models/funcion.model';

@Component({
  imports: [RouterLink],
  selector: 'app-detalle-pelicula',
  styleUrl: './detalle-pelicula.css',
  templateUrl: './detalle-pelicula.html',
})
export class DetallePelicula {
  pelicula?: Pelicula;
  funciones: Funcion[] = [];

  //Traigo el servicio que contiene las películas y
  //el servicio que contiene las funciones
  constructor(
    private route: ActivatedRoute,
    private peliculaService: PeliculaService,
    private funcionService: FuncionService,
    private router: Router,
  ) {}

  ngOnInit(): void {
    //Obtengo el id de la película desde la URL por medio de ActivatedRoute
    const id = Number(this.route.snapshot.paramMap.get('id'));
    if (id) {
      //Obtengo la película y sus funciones
      this.pelicula = this.peliculaService.obtenerPeliculaPorId(id);
      this.funciones = this.funcionService.obtenerFuncionesPorPelicula(id);
    }
  }

  seleccionarFuncion(funcion: Funcion) {
    this.router.navigate(['/compra', funcion.id]);
  }
}

<div class="detalle-pelicula d-flex justify-content-center">
  @if (pelicula) {
    <div class="m-3 p-3 border border-warning">
      <img [src]="pelicula.imagen" [alt]="pelicula.nombre" />

      <h1>{{ pelicula.nombre }}</h1>

      <p>{{ pelicula.sinopsis }}</p>

      <p>Duración: {{ pelicula.duracion }} minutos</p>

      <p>Género: {{ pelicula.genero }}</p>

      <p>Edad: +{{ pelicula.clasificacionEdad }}</p>

      <button routerLink="/peliculas">Volver a películas</button>
    </div>
  }
</div>

@if (funciones.length > 0) {
  <h3 class="d-flex justify-content-center m-4">Funciones</h3>
  @for (funcion of funciones; track $index) {
    <div class="detalle-pelicula d-flex justify-content-center m-3">
      <div>
        <p>Fecha: {{ funcion.fecha }}</p>
        <p>Hora: {{ funcion.hora }}</p>
        <p>Sala: {{ funcion.salaId }}</p>
      </div>
      <div>
        <button (click)="seleccionarFuncion(funcion)" class="m-3">Seleccionar</button>
      </div>
    </div>
  }
} @else {
  <div class="d-flex justify-content-center">
    <p>No hay funciones disponibles.</p>
  </div>
}
